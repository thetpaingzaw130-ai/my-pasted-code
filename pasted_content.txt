#!/usr/bin/env python3
"""Create a local inventory of files on a Termux-accessible storage path.

This script never uploads, sends, or transmits files. It only writes a CSV
inventory locally and skips Android protected directories.
"""

import argparse
import csv
import os
from pathlib import Path


def scan_files(root: Path, output: Path, max_size_mb: int):
    max_size = max_size_mb * 1024 * 1024
    output.parent.mkdir(parents=True, exist_ok=True)
    count = 0

    with output.open("w", newline="", encoding="utf-8") as handle:
        writer = csv.writer(handle)
        writer.writerow(["path", "size_bytes", "modified_unix"])

        for current_root, dirs, files in os.walk(root, topdown=True):
            # Avoid Android app-private/protected locations.
            dirs[:] = [
                name for name in dirs
                if name not in {"Android"}
            ]

            for filename in files:
                path = Path(current_root) / filename
                try:
                    stat = path.stat()
                    if stat.st_size <= max_size:
                        writer.writerow([str(path), stat.st_size, int(stat.st_mtime)])
                        count += 1
                except (OSError, PermissionError):
                    continue

    return count


def main():
    parser = argparse.ArgumentParser(
        description="Create a local-only CSV inventory of accessible files."
    )
    parser.add_argument(
        "root",
        nargs="?",
        default="/sdcard",
        help="Directory to scan (default: /sdcard)",
    )
    parser.add_argument(
        "-o",
        "--output",
        default="file_inventory.csv",
        help="Local CSV output path (default: file_inventory.csv)",
    )
    parser.add_argument(
        "--max-size-mb",
        type=int,
        default=48,
        help="Only include files up to this size (default: 48 MB)",
    )
    args = parser.parse_args()

    root = Path(args.root).expanduser()
    if not root.is_dir():
        raise SystemExit(f"Directory not found or inaccessible: {root}")
    if args.max_size_mb < 0:
        raise SystemExit("--max-size-mb must be zero or greater")

    count = scan_files(root, Path(args.output).expanduser(), args.max_size_mb)
    print(f"Wrote {count} file entries to {Path(args.output).expanduser()}")


if __name__ == "__main__":
    main()
